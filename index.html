from aiogram import Bot, Dispatcher
import asyncio

bot = Bot(token="YOUR_TOKEN")
dp = Dispatcher()

@dp.message()
async def start(message):
    await message.answer("Automation works")

asyncio.run(dp.start_polling(bot))
